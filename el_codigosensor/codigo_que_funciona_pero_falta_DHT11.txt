#include <WiFi.h>
#include <WebServer.h>
#include <DHT.h>

#define DHTPIN 13          // Pin donde está conectado el sensor DHT
#define DHTTYPE DHT11     // Tipo de sensor (puedes cambiar a DHT22 si es ese el que usas)

const char* ssid = "mai";  // Cambia por el nombre de tu red WiFi
const char* password = "maiguelb"; // Cambia por la contraseña de tu red WiFi

WebServer server(80);  // Crear un servidor HTTP en el puerto 80
DHT dht(DHTPIN, DHTTYPE);  // Crear el objeto DHT

void setup() {
  Serial.begin(115200);
  WiFi.begin(ssid, password);  // Conectar a la red WiFi
  
  while (WiFi.status() != WL_CONNECTED) {
    delay(1000);
    Serial.println("Conectando a WiFi...");
  }

  Serial.println("Conectado a WiFi!");
  Serial.print("IP Address: ");
  Serial.println(WiFi.localIP());  // Mostrar la IP del ESP32

  dht.begin();  // Inicializar el sensor DHT

  // Definir la ruta "/read" para obtener la lectura del sensor
  server.on("/read", HTTP_GET, []() {
    float temperatura = dht.readTemperature();  // Leer la temperatura
    float humedad = dht.readHumidity();        // Leer la humedad

    // Comprobar si la lectura fue exitosa
    if (isnan(temperatura) || isnan(humedad)) {
      server.send(500, "text/plain", "Error al leer el sensor DHT.");
    } else {
      // Enviar los datos como respuesta (en formato texto)
      String response = "Temperatura: " + String(temperatura) + "°C\n";
      response += "Humedad: " + String(humedad) + "%";
      server.send(200, "text/plain", response);
    }
  });

  // Iniciar el servidor
  server.begin();
}

void loop() {
  server.handleClient();  // Manejar las solicitudes HTTP entrantes
}
